int reverse(int n)
{

	int rev=0;
	while(n>0)
	{
		int rem=n%10;
		rev=rev*10+rem;
		n=n/10;
	}
	return rev;
}

long int dtob(int n)
{

    long int r=0,s=1;

    {
        int rem=n%2;
        r=rem*s+r;

        n=n/2;
    }
     return r;
    
}
