#include <stdio.h>

int reverse(int n)
{

	int rev=0;
	while(n>0)
	{
		int rem=n%10;
		rev=rev*10+rem;
		n=n/10;
	}
	return rev;
}
void count(int n)
{
	int even_count=0;
	int odd_count=0;
	
	while(n>0)
	{
		int rem=n%10;
		if(rem%2==0)
		{
			even_count++;
		}
		else
		{
			odd_count++;
		}
		n=n/10;
	}
	printf("2.even count:%d\n odd count:%d\n",even_count,odd_count);
}
void evnodd(int n)

		{
			ev=ev*10+rem;
		}

			od=od*10+rem;
		}

}

int summation(int n)
{
int sum=0;
while(n>0)
{
    sum=sum+n;
    n=n-1;
}
return sum;

}
void summation_of_digits(int n)
{
    int sum=0;
    int rem=0;
    while(n>0)
    {
        rem=n%10;
        sum=sum+summation(rem);
        n=n/10;
        
    }
    printf("7.summation of digits:%d\n",sum);
}
int fact(int n)
{
int mul=1;
while(n>0)
{
    mul=mul*n;
    n=n-1;
}
return mul;

}
void factorial_of_digits(int n)
{
    int sum=0;
    int rem=0;
    while(n>0)
    {
        rem=n%10;
        sum=sum+fact(rem);
        n=n/10;
        
    }
    printf("8.factorial of digits:%d\n",sum);
}
int power(int b,int e)
{
    int p=1;
    for(int i=0;i<e;i++)
    {
        p=p*b;
    }
    return p;
}
int square(int n)
{
    int sum=0;
    while(n>0)
    {
        int rem=n%10;
        
        sum=sum+power(rem,2);
        n=n/10;
    }
    return sum;
}
int power6(int n)
{
    int sum=0;
    while(n>0)
    {
        int rem=n%10;
        
        sum=sum+power(rem,6);
        n=n/10;
    }
    return sum;
}
